cmake_minimum_required(VERSION 3.22)
project(Nebula VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(NEBULA_BUILD_EDITOR "Build the Qt editor feasibility prototype" ON)

add_library(nebula_core src/core/document.cpp src/core/database.cpp src/core/wiki.cpp)
target_include_directories(nebula_core PUBLIC include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(nebula_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE nebula_core)
    add_test(NAME core COMMAND core_tests)
    if(UNIX)
        find_program(BASH_EXECUTABLE bash REQUIRED)
        add_test(NAME launcher COMMAND "${BASH_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/launcher_tests.sh")
        set_tests_properties(launcher PROPERTIES TIMEOUT 15
            ENVIRONMENT "NEBULA_TEST_TMPDIR=${CMAKE_CURRENT_BINARY_DIR}")
    endif()
endif()

if(NEBULA_BUILD_EDITOR)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
    find_package(SQLite3 3.33 REQUIRED)
    add_library(nebula_format src/format/codec.cpp)
    target_include_directories(nebula_format PUBLIC include)
    target_link_libraries(nebula_format PUBLIC nebula_core PRIVATE Qt6::Core)
    add_library(nebula_files src/attachments/files.cpp src/attachments/manifest.cpp)
    target_include_directories(nebula_files PUBLIC include)
    target_link_libraries(nebula_files PRIVATE Qt6::Core)
    add_library(nebula_storage src/storage/workspace.cpp src/storage/records.cpp src/storage/backup.cpp src/storage/attachments.cpp src/storage/search.cpp src/storage/link_query.cpp)
    target_include_directories(nebula_storage PUBLIC include)
    target_link_libraries(nebula_storage PUBLIC nebula_core nebula_format nebula_files PRIVATE SQLite::SQLite3 Qt6::Core)
    add_library(nebula_interchange src/interchange/markdown.cpp src/interchange/json.cpp)
    target_include_directories(nebula_interchange PUBLIC include)
    target_link_libraries(nebula_interchange PUBLIC nebula_storage PRIVATE SQLite::SQLite3 Qt6::Core)
    add_library(nebula_application src/app/page_session.cpp)
    target_link_libraries(nebula_application PUBLIC nebula_storage)
    add_executable(nebula_workspace src/cli/workspace_main.cpp)
    target_link_libraries(nebula_workspace PRIVATE nebula_interchange)
    add_library(nebula_editor
        src/editor/page_editor.cpp
        src/editor/block_widgets.cpp
        include/nebula/editor/page_editor.hpp
        include/nebula/editor/block_widgets.hpp)
    set_target_properties(nebula_editor PROPERTIES AUTOMOC ON)
    target_link_libraries(nebula_editor PUBLIC nebula_core Qt6::Widgets PRIVATE nebula_format)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(nebula_editor PRIVATE -Wall -Wextra -Wpedantic)
    endif()

    add_library(nebula_desktop
        src/app/trash_controls.cpp src/app/workspace_trash.cpp
        src/app/workspace_dialog.cpp src/app/prototype_window.cpp
        src/app/workspace_window.cpp include/nebula/app/workspace_window.hpp
        include/nebula/app/workspace_dialog.hpp include/nebula/app/prototype_window.hpp)
    set_target_properties(nebula_desktop PROPERTIES AUTOMOC ON)
    target_link_libraries(nebula_desktop PUBLIC nebula_editor nebula_application nebula_interchange)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(nebula_storage PRIVATE -Wall -Wextra -Wpedantic)
        target_compile_options(nebula_interchange PRIVATE -Wall -Wextra -Wpedantic)
        target_compile_options(nebula_files PRIVATE -Wall -Wextra -Wpedantic)
        target_compile_options(nebula_format PRIVATE -Wall -Wextra -Wpedantic)
        target_compile_options(nebula_desktop PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_executable(nebula_prototype src/prototype/main.cpp)
    target_link_libraries(nebula_prototype PRIVATE nebula_desktop)

    if(BUILD_TESTING)
        find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
        add_executable(storage_tests tests/storage_tests.cpp)
        add_executable(recovery_writer tests/recovery_writer.cpp)
        target_link_libraries(recovery_writer PRIVATE SQLite::SQLite3)
        set_target_properties(storage_tests PROPERTIES AUTOMOC ON)
        target_link_libraries(storage_tests PRIVATE nebula_application Qt6::Test SQLite::SQLite3)
        add_dependencies(storage_tests nebula_workspace recovery_writer)
        add_test(NAME storage COMMAND storage_tests)
        add_executable(interchange_tests tests/interchange_tests.cpp)
        set_target_properties(interchange_tests PROPERTIES AUTOMOC ON)
        target_link_libraries(interchange_tests PRIVATE nebula_interchange Qt6::Test SQLite::SQLite3)
        add_test(NAME interchange COMMAND interchange_tests)
        add_executable(editor_tests tests/editor_tests.cpp)
        set_target_properties(editor_tests PROPERTIES AUTOMOC ON)
        target_link_libraries(editor_tests PRIVATE nebula_editor Qt6::Test)
        add_test(NAME editor COMMAND editor_tests)
        set_tests_properties(editor PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
        add_executable(desktop_tests tests/desktop_tests.cpp)
        set_target_properties(desktop_tests PROPERTIES AUTOMOC ON)
        target_link_libraries(desktop_tests PRIVATE nebula_desktop Qt6::Test SQLite::SQLite3)
        add_dependencies(desktop_tests nebula_prototype)
        add_test(NAME desktop COMMAND desktop_tests)
        set_tests_properties(desktop PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
        add_executable(trash_safety_tests tests/trash_safety_tests.cpp)
        set_target_properties(trash_safety_tests PROPERTIES AUTOMOC ON)
        target_link_libraries(trash_safety_tests PRIVATE nebula_desktop Qt6::Test SQLite::SQLite3)
        add_test(NAME trash_safety COMMAND trash_safety_tests)
        set_tests_properties(trash_safety PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
        add_test(NAME prototype_smoke COMMAND nebula_prototype --smoke-test)
        set_tests_properties(prototype_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
    endif()
endif()
